Артефакт: фрагмент контракта AsyncAPI 3.0 (обезличен, по мотивам проекта 3)

asyncapi: 3.0.0
info:
  title: "Событие изменения статуса контракта"
channels:
  contract.banking.v1:
    address: contract.banking.v1
    messages:
      contractStatusChanged:
        payload:
          type: object
          required: [event_id, contract_id, status, occurred_at]
          properties:
            event_id: { type: string, format: uuid }            # дедупликация
            contract_id: { type: string }
            status: { type: string, enum: [ACTIVE, DELINQUENT, CLOSED] }
            occurred_at: { type: string, format: date-time }
x-error-model:
  validation_error:    { code: 422, retry: false }
  duplicate_event:    { treated_as: success }               # идемпотентность
  service_unavailable: { code: 503, retry: true, backoff: exponential }

Синтетический пример: показывает, как фиксирую идемпотентность, статусные модели и единую error-model прямо в контракте.
